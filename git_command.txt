Command	Purpose
git init	Initialize a new Git repository
git clone <repo-url>	Clone an existing repository
git --version	Check Git version
git config --global user.name "Your Name"	Set Git username
git config --global user.email "you@example.com"	Set Git email
git config --list	View Git configuration
git status	Show the current status of files
git log	View commit history
git log --oneline	View commit history (short format)
git reflog	View all HEAD changes
git add .	Stage all changes
git add <file>	Stage a specific file
git restore --staged <file>	Unstage a file
git commit -m "message"	Commit staged changes
git commit --amend	Modify the last commit
git branch	List local branches
git branch <branch-name>	Create a new branch
git switch <branch-name>	Switch to a branch
git switch -c <branch-name>	Create and switch to a new branch
git checkout -b <branch-name>	Create and switch (older syntax)
git branch -d <branch-name>	Delete a branch
git branch -D <branch-name>	Force delete a branch
git remote -v	View remote repositories
git remote add origin <repo-url>	Add a remote repository
git remote set-url origin <repo-url> If you want to point origin to a different GitHub repository
git remote remove origin	Remove a remote repository
git push origin main	Push to the main branch
git push -u origin main	Push and set upstream
git pull origin main	Pull latest changes
git fetch	Fetch changes without merging
git merge <branch>	Merge a branch into the current branch
git rebase <branch>	Rebase the current branch onto another branch
git restore <file>	Discard changes in a file
git reset --soft HEAD~1	Undo last commit (keep staged changes)
git reset HEAD~1	Undo last commit (keep files)
git reset --hard HEAD~1	Undo last commit and discard changes
git stash	Save uncommitted changes
git stash list	List stashes
git stash pop	Apply and remove the latest stash
git stash apply	Apply the latest stash without removing it
git diff	Show unstaged changes
git diff --staged	Show staged changes
git tag	List tags
git tag v1.0	Create a tag
git push origin --tags	Push tags to remote
